async function deleteInvoice(invoiceNo) { if (!confirm(`Are you sure you want to delete Invoice ${invoiceNo}?`)) return; const apiUrl = localStorage.getItem('gst_sheet_api_url'); try { const res = await fetch(apiUrl, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: JSON.stringify({ action: 'deleteInvoice', invoiceNo }) }); const result = await res.json(); if (result.status === 'success') { alert(`Invoice ${invoiceNo} deleted successfully.`); // CRITICAL FIX: If we were currently editing this exact invoice, reset the form! const currentEditingNo = document.getElementById('invNo').value; if (currentEditingNo === invoiceNo) { resetForm(); } loadInvoices(); } else { throw new Error(result.message); } } catch (err) { alert('Failed to delete invoice: ' + err.message); } }
